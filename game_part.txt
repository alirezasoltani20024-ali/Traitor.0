function drawPlayer(p){
  if(!p.alive && !p.corpse)return;
  ctx.save();ctx.translate(p.x,p.y);
  if(p.corpse){
    ctx.rotate(-0.18);
    ctx.fillStyle='#8b3f4b';ctx.strokeStyle='#3b2026';ctx.lineWidth=3;
    ctx.beginPath();ctx.ellipse(0,10,25,13,0,0,Math.PI*2);ctx.fill();ctx.stroke();
    ctx.fillStyle=p.color||'#fb041c';ctx.beginPath();ctx.arc(-8,-3,12,0,Math.PI*2);ctx.fill();ctx.stroke();
    ctx.fillStyle='#c9f1ff';ctx.beginPath();ctx.roundRect(-5,-7,16,9,5);ctx.fill();
    ctx.fillStyle='#16364b';ctx.beginPath();ctx.roundRect(-3,-5,12,6,3);ctx.fill();
    ctx.fillStyle='#fff';ctx.font='13px Tahoma';ctx.textAlign='center';ctx.fillText(p.name,0,34);
    ctx.restore();return;
  }

  // هر بازیکن از خانه دقیق خودش در Sprite Sheet می‌آید؛ رنگ و اسکین با index عددی هماهنگ‌اند.
  const colorIndex=Number.isInteger(p.colorIndex)?Math.max(0,Math.min(19,p.colorIndex)):
    Math.max(0,ART_COLOR_INDEX.get(String(p.color||'').toLowerCase())??0);
  const shapeIndex=Number.isInteger(p.shapeIndex)?Math.max(0,Math.min(10,p.shapeIndex)):
    Math.max(0,FREE_SKINS.findIndex(x=>x[1]===(p.shape||'classic')));
  const sx=shapeIndex*ART_CELL, sy=colorIndex*ART_CELL;

  // سایه نرم زیر شخصیت تا رنگ‌های روشن (سفید/کرم/یاسی) هم کاملاً خوانا بمانند.
  ctx.fillStyle='rgba(0,0,0,.28)';
  ctx.beginPath();ctx.ellipse(0,31,29,8,0,0,Math.PI*2);ctx.fill();

  if(CHARACTER_ART.complete&&CHARACTER_ART.naturalWidth){
    ctx.save();
    ctx.imageSmoothingEnabled=true;
    ctx.imageSmoothingQuality='high';
    ctx.shadowColor='rgba(0,0,0,.22)';ctx.shadowBlur=5;ctx.shadowOffsetY=2;
    ctx.drawImage(CHARACTER_ART,sx,sy,ART_CELL,ART_CELL,-46,-46,92,92);
    ctx.restore();
  }else{
    ctx.fillStyle=p.color||FREE_COLORS[colorIndex][1];ctx.beginPath();ctx.roundRect(-22,-26,44,52,18);ctx.fill();
  }

  ctx.fillStyle='#fff';ctx.font='14px Tahoma';ctx.textAlign='center';ctx.fillText(p.name,0,-54);
  if(performance.now()<roleTimer&&p.id===myId){
    ctx.fillStyle=role==='infiltrator'?'#d92742':'#267ee8';ctx.font='bold 22px Tahoma';
    ctx.fillText(role==='infiltrator'?'خائن':'خدمه',0,-75);
  }
  if(role==='infiltrator'&&partners.includes(p.name)&&p.id!==myId){
    ctx.fillStyle='#e32643';ctx.font='bold 14px Tahoma';ctx.fillText(p.name,0,-68)
  }
  ctx.restore();
}
